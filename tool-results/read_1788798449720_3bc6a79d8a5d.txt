   102→      qc.invalidateQueries({ queryKey: ['emails'] })
   103→      qc.invalidateQueries({ queryKey: qk.dashboard() })

   105→        toast({

   108→        })

   314→// ---------- Notification preferences (per-category) ----------
   315→export type NotificationPrefChannel = 'in_app' | 'web' | 'push'
   316→export interface NotificationPrefsData {
   317→  channels: {
   318→    channel: NotificationPrefChannel
   319→    preferences: { categoryId: string | '__global__'; enabled: boolean }[]
   320→  }[]
   321→}
   322→export function useNotificationPreferences() {
   323→  return useQuery({
   324→    queryKey: ['notification-preferences'],
   325→    queryFn: () => api.notificationPreferences.list() as Promise<NotificationPrefsData>,
   326→    placeholderData: (prev) => prev,
   327→  })
   328→}
   329→export function useUpdateNotificationPreference() {
   330→  const qc = useQueryClient()
   331→  const { toast } = useToast()
   332→  return useMutation({
   333→    mutationFn: api.notificationPreferences.update,
   334→    // Optimistic update so the Switch animates instantly — the refetched
   335→    // data on settle reconciles any drift.
   336→    onMutate: async ({ channel, categoryId, enabled }) => {
   337→      await qc.cancelQueries({ queryKey: ['notification-preferences'] })
   338→      const prev = qc.getQueryData<NotificationPrefsData>(['notification-preferences'])
   339→      const key = categoryId ?? '__global__'
   340→      if (prev) {
   341→        const next: NotificationPrefsData = {
   342→          ...prev,
   343→          channels: prev.channels.map((c) =>
   344→            c.channel === channel
   345→              ? {
   346→                  ...c,
   347→                  preferences: c.preferences.map((p) =>
   348→                    p.categoryId === key ? { ...p, enabled } : p,
   349→                  ),
   350→                }
   351→              : c,
   352→          ),
   353→        }
   354→        qc.setQueryData<NotificationPrefsData>(['notification-preferences'], next)
   355→      }
   356→      return { prev }
   357→    },
   358→    onError: (_err, _vars, ctx) => {
   359→      if (ctx?.prev) {
   360→        qc.setQueryData<NotificationPrefsData>(['notification-preferences'], ctx.prev)
   361→      }
   362→      toast({ title: 'Failed to update preference', variant: 'destructive' })
   363→    },
   364→    onSuccess: () => {
   365→      toast({ title: 'Notification preference updated' })
   366→    },
   367→    onSettled: () => {
   368→      qc.invalidateQueries({ queryKey: ['notification-preferences'] })
   369→    },
   370→  })
   371→}

   509→    mutationFn: ({ id, title, archived }: { id: string; title?: string; archived?: boolean }) =>
   510→      api.assistant.updateConversation(id, { ...(title !== undefined ? { title } : {}), ...(archived !== undefined ? { archived } : {}) }),
   511→    onSuccess: (_d, { title, archived }) => {
   512→      qc.invalidateQueries({ queryKey: qk.assistantConversations() })
   513→      // Prefer the more-specific toast when both fields were supplied.
   514→      if (archived !== undefined) {
   515→        toast({ title: archived ? 'Conversation archived' : 'Conversation unarchived' })
   516→      } else if (title !== undefined) {
   517→        toast({ title: 'Conversation renamed' })
   518→      }
   519→    },
   520→  })

   591→  | 'read'

   662→// ---------- Weekly digest ----------
   663→// On-demand AI weekly digest. POSTs to /api/assistant/digest which gathers
   664→// real 7-day mailbox stats and synthesizes a structured digest via the LLM
   665→// (with a deterministic fallback if the LLM call fails). The digest is NOT
   666→// persisted — every call regenerates it against the latest state so the user
   667→// always sees fresh data. Errors surface a destructive toast; the calling
   668→// component reads mutation.isPending for the loading state and mutation.data
   669→// for the digest payload.
   670→export function useWeeklyDigest() {
   671→  const { toast } = useToast()
   672→  return useMutation({
   673→    mutationFn: async (): Promise<WeeklyDigest> => {
   674→      const res = await fetch('/api/assistant/digest', { method: 'POST' })
   675→      if (!res.ok) {
   676→        const body = await res.json().catch(() => null)
   677→        const msg = body && typeof body === 'object' && 'error' in body
   678→          ? String((body as { error: string }).error)
   679→          : `Failed to generate digest (${res.status})`
   680→        throw new Error(msg)
   681→      }
   682→      return (await res.json()) as WeeklyDigest
   683→    },
   684→    onError: () => {
   685→      toast({ title: 'Failed to generate digest', variant: 'destructive' })
   686→    },
   687→  })
   688→}
