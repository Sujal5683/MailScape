    79→// ---------------------------------------------------------------------------
    80→// Exported component — signature preserved for inbox-view / search-view /
    81→// organized/category-detail. Two new optional props added (default no-op).
    82→// ---------------------------------------------------------------------------
    83→export function EmailDetail({
    84→  emailId,
    85→  onBack,
    86→  onSelectThreadEmail,
    87→  onNavigate,
    88→}: {
    89→  emailId: string | null
    90→  onBack?: () => void
    91→  onSelectThreadEmail?: (id: string) => void
    92→  onNavigate?: (direction: 'prev' | 'next') => void
    93→}) {
    94→  const { data: email, isLoading, error, refetch } = useEmail(emailId)
    95→
    96→  if (!emailId) {
    97→    return (
    98→      <div className="flex h-full items-center justify-center p-10">
    99→        <div className="text-center">
   100→          <Mail className="mx-auto h-10 w-10 text-muted-foreground/40" />
   101→          <p className="mt-3 text-sm font-medium">Select an email to read</p>
   102→          <p className="mt-1 text-xs text-muted-foreground">
   103→            Your messages appear here with full context.
   104→          </p>
   105→        </div>
   106→      </div>
   107→    )
   108→  }
   109→
   110→  if (isLoading) {
   111→    return (
   112→      <div className="space-y-3">
   113→        <EmailHeaderSkeleton />
   114→        <SleekSeparator />
   115→        <EmailBodySkeleton />
   116→      </div>
   117→    )
   118→  }
   119→  if (error) {
   120→    return (
   121→      <ErrorState
   122→        title="Couldn't load email"
   123→        description={error.message}
   124→        onRetry={() => refetch()}
   125→      />
   126→    )
   127→  }
   128→  if (!email) return <ErrorState title="Email not found" />
   129→
   130→  return (
   131→    <EmailDetailContent
   132→      emailId={emailId}
   133→      onBack={onBack}
   134→      onSelectThreadEmail={onSelectThreadEmail}
   135→      onNavigate={onNavigate}
   136→    />
   137→  )
   138→}

   143→function EmailDetailContent({

   145→  onBack,
   146→  onSelectThreadEmail,
   147→  onNavigate,
   148→}: {
   149→  emailId: string

   151→  onSelectThreadEmail?: (id: string) => void
   152→  onNavigate?: (direction: 'prev' | 'next') => void
   153→}) {
   154→  const { data: email } = useEmail(emailId)

   156→  const { data: classification } = useClassificationResults(emailId)
   157→  const { data: categories } = useCategories()
   158→
   159→  const markRead = useMarkRead()
   160→  const star = useStarEmail()
   161→  const important = useMarkImportant()
   162→  const unsnooze = useSnoozeEmail()

   164→  const setComposeOpen = useUIStore((s) => s.setComposeOpen)
   165→  const smartReplies = useSmartReplies()
   166→
   167→  const [threadOpen, setThreadOpen] = useState(true)
   168→  const [previewAttachment, setPreviewAttachment] = useState<EmailAttachmentMeta | null>(null)
   169→  const [snoozeOpen, setSnoozeOpen] = useState(false)
   170→  const [smartRepliesOpen, setSmartRepliesOpen] = useState(false)
   171→

   173→  // user explicitly opts in per email (and stale results from the previous

   176→  // `react-hooks/set-state-in-effect` lint rule.

   178→  const [prevEmailId, setPrevEmailId] = useState(emailId)

   180→    setPrevEmailId(emailId)
   181→    setSmartRepliesOpen(false)
   182→  }
   183→
   184→  // Keyboard shortcuts: j/k navigate prev/next (if wired), ? shows the help toast.
   185→  // Guarded against input/textarea/contenteditable so it never hijacks typing.
   186→  useEffect(() => {
   187→    function handler(e: KeyboardEvent) {
   188→      const t = e.target as HTMLElement | null
   189→      const tag = t?.tagName
   190→      if (tag === 'INPUT' || tag === 'TEXTAREA' || t?.isContentEditable) return

   192→      if (e.key === 'j') {
   193→        onNavigate?.('next')
   194→      } else if (e.key === 'k') {
   195→        onNavigate?.('prev')
   196→      } else if (e.key === '?') {
   197→        toast.info('Keyboard shortcuts', {
   198→          description: 'j / k — next / previous email · s — star · e — archive · r — reply',
   199→        })
   200→      }
   201→    }
   202→    window.addEventListener('keydown', handler)
   203→    return () => window.removeEventListener('keydown', handler)
   204→  }, [onNavigate])
   205→
   206→  const wordCount = useMemo(() => {
   207→    const text = email?.bodyText ?? email?.bodyHtmlSanitized?.replace(/<[^>]+>/g, ' ') ?? ''
   208→    return text.trim().split(/\s+/).filter(Boolean).length
   209→  }, [email?.bodyText, email?.bodyHtmlSanitized])
   210→
   211→  if (!email) return null
   212→
   213→  const cat = email.categories[0]
   214→  const catColorClass = cat ? colorClass(cat.color) : 'cat-slate'
   215→
   216→  const domain = email.fromEmail.split('@')[1]?.toLowerCase() ?? ''
   217→  const isInstitutional = domain === 'iitjammu.ac.in'
   218→
   219→  const others = (thread?.emails ?? []).filter((m) => m.id !== email.id)
   220→
   221→  const handleGmailOpen = () => {
   222→    window.open(email.gmailUrl, '_blank', 'noopener,noreferrer')
   223→  }
   224→  const handleArchive = () => toast.info('Archive is cosmetic in this build', {
   225→    description: 'Connected to Gmail, this would remove from the inbox view.',
   226→  })
   227→  const handleReply = () => {
   228→    setComposeOpen(true)
   229→    toast.info('Reply prefill loaded in compose', {
   230→      description: `Replying to ${email.fromName ?? email.fromEmail}`,
   231→    })
   232→  }
   233→  const handleReplyAll = () => {
   234→    setComposeOpen(true)
   235→    toast.info('Reply-all prefill loaded in compose')
   236→  }
   237→  const handleForward = () => {
   238→    setComposeOpen(true)
   239→    toast.info('Forward prefill loaded in compose')
   240→  }
   241→  const handleAskAI = () => navigate('assistant', { contextEmailId: email.id })
   242→  // Open the Smart Replies panel and immediately kick off generation. If a
   243→  // request is already in-flight (e.g. the user double-clicked), we skip the
   244→  // second trigger — the panel will reflect the pending state from the first.
   245→  const handleSmartReplies = () => {
   246→    setSmartRepliesOpen(true)
   247→    if (!smartReplies.isPending) {
   248→      smartReplies.mutate(email.id)
   249→    }
   250→  }
   251→  const handleAddDeadline = () =>
   252→    toast.info('Add to deadlines', { description: 'Deadline extraction is cosmetic in this build.' })
   253→  const handleCreateRule = () => {
   254→    navigate('rules')
   255→    toast.info('Create rule from sender', {
   256→      description: `Prefilled for ${email.fromEmail}`,
   257→    })
   258→  }
   259→  const handleMoveToSection = (name: string) =>
   260→    toast.info('Move to section', { description: `Moving to "${name}" is cosmetic in this build.` })
   261→
   262→  return (
   263→    <div className="flex h-full flex-col">
   264→      {/* ===== Action toolbar (pinned at top of detail pane; wraps on mobile) ===== */}
   265→      <div className="shrink-0 border-b bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/80">

   267→          {onBack && (
   268→            <Button variant="ghost" size="sm" onClick={onBack} className="shrink-0">
   269→              <ArrowLeft className="h-4 w-4" />
   270→              <span className="hidden sm:inline">Back</span>
   271→            </Button>
   272→          )}
   273→          {onBack && <Separator orientation="vertical" className="mx-1 h-6" />}
   274→
   275→          {/* Group 1: status toggles */}
   276→          <ToolbarButton
   277→            icon={Star}
   278→            label="Star"
   279→            onClick={() => star.mutate({ id: email.id, starred: !email.flags.isStarred })}
   280→            active={email.flags.isStarred}
   281→            activeClass="text-warning"
   282→          />
   283→          <ToolbarButton
   284→            icon={MailOpen}
   285→            label={email.flags.isRead ? 'Mark unread' : 'Mark read'}
   286→            onClick={() => markRead.mutate({ id: email.id, read: !email.flags.isRead })}
   287→            alternateIcon={email.flags.isRead ? Mail : MailOpen}
   288→          />
   289→          <ToolbarButton
   290→            icon={ShieldCheck}
   291→            label={email.flags.isImportant ? 'Remove importance' : 'Mark important'}
   292→            onClick={() => important.mutate({ id: email.id, important: !email.flags.isImportant })}
   293→            active={email.flags.isImportant}
   294→            activeClass="text-warning"
   295→          />
   296→
   297→          <Separator orientation="vertical" className="mx-1 h-6 hidden sm:block" />
   298→
   299→          {/* Group 2: actions */}
   300→          <ToolbarButton icon={Archive} label="Archive" onClick={handleArchive} />
   301→          <ToolbarButton
   302→            icon={Clock}
   303→            label={isActivelySnoozed(email.snoozedUntil) ? 'Reschedule snooze' : 'Snooze'}
   304→            onClick={() => setSnoozeOpen(true)}
   305→            active={isActivelySnoozed(email.snoozedUntil)}
   306→            activeClass="text-accent-foreground bg-accent"
   307→          />
   308→          <ToolbarButton icon={Reply} label="Reply" onClick={handleReply} />
   309→          <ToolbarButton icon={ReplyAll} label="Reply all" onClick={handleReplyAll} />
   310→          <ToolbarButton icon={Forward} label="Forward" onClick={handleForward} />

   312→            icon={Wand2}
   313→            label="Smart replies"
   314→            onClick={handleSmartReplies}
   315→            active={smartRepliesOpen}
   316→            activeClass="text-primary bg-primary/10"
   317→          />
   318→
   319→          <Separator orientation="vertical" className="mx-1 h-6 hidden sm:block" />
   320→
   321→          {/* Group 3: external / AI */}
   322→          <ToolbarButton icon={ExternalLink} label="Open in Gmail" onClick={handleGmailOpen} />
   323→          <ToolbarButton
   324→            icon={Sparkles}
   325→            label="Ask AI"
   326→            onClick={handleAskAI}
   327→            activeClass="text-primary"
   328→          />
   329→
   330→          {onNavigate && (
   331→            <div className="ml-auto hidden items-center gap-1 md:flex">
   332→              <ToolbarButton icon={ChevronRight} label="Previous" onClick={() => onNavigate('prev')} rotate />
   333→              <ToolbarButton icon={ChevronRight} label="Next" onClick={() => onNavigate('next')} />
   334→            </div>
   335→          )}
   336→        </div>
   337→      </div>
   338→
   339→      {/* ===== Scrollable reading column ===== */}
   340→      <ScrollArea className="min-h-0 flex-1">
   341→        <div className="mx-auto w-full max-w-3xl px-4 py-5 pb-20 md:px-6 md:py-6 md:pb-6">
   342→          {/* Snooze banner — only when actively snoozed */}
   343→          {isActivelySnoozed(email.snoozedUntil) && (
   344→            <div className="mb-4 flex flex-wrap items-center gap-2 rounded-lg border border-accent bg-accent/40 px-3 py-2 text-xs">
   345→              <Clock className="h-3.5 w-3.5 text-accent-foreground" />
   346→              <span className="font-medium text-accent-foreground">Snoozed until {formatDateTime(email.snoozedUntil ?? null)}</span>
   347→              <span className="text-muted-foreground">
   348→                It&apos;s hidden from your default inbox until then.
   349→              </span>

   351→                <Button variant="ghost" size="sm" className="h-7 px-2 text-xs" onClick={() => setSnoozeOpen(true)}>
   352→                  Reschedule
   353→                </Button>
   354→                <Button
   355→                  variant="ghost"
   356→                  size="sm"
   357→                  className="h-7 px-2 text-xs"
   358→                  onClick={() => unsnooze.mutate({ id: email.id, until: null })}
   359→                  disabled={unsnooze.isPending}
   360→                >
   361→                  Unsnooze
   362→                </Button>
   363→              </div>
   364→            </div>
   365→          )}
   366→
   367→          {/* Subject + category */}
   368→          <div className="mb-3">
   369→            <h1 className="text-lg font-semibold leading-tight tracking-tight md:text-xl">
   370→              {email.subject ?? '(no subject)'}
   371→            </h1>
   372→            {cat && (
   373→              <button
   374→                onClick={() => navigate('organized', { contextCategoryId: cat.id })}
   375→                className={cn(
   376→                  'mt-2 inline-flex items-center gap-1.5 rounded-full border px-2.5 py-1 text-xs font-medium cat-bg-soft cat-text cat-border-soft transition-opacity hover:opacity-80',
   377→                  catColorClass,
   378→                )}
   379→              >
   380→                <CategoryIcon icon={cat.icon} color={cat.color} className="h-3.5 w-3.5" />
   381→                {cat.name}
   382→                <span className="text-[10px] opacity-70">· {cat.source.replace(/_/g, ' ')}</span>
   383→              </button>
   384→            )}
   385→          </div>
   386→
   387→          {/* Polished sender card */}
   388→          <div className="mb-4 rounded-xl border bg-card p-4">
   389→            <div className="flex items-start gap-3">
   390→              <div
   391→                className={cn(
   392→                  'flex h-12 w-12 shrink-0 items-center justify-center rounded-full text-base font-semibold ring-1 ring-inset cat-bg-soft cat-text cat-border-soft',
   393→                  catColorClass,
   394→                )}
   395→                aria-hidden
   396→              >
   397→                {(email.fromName ?? email.fromEmail).charAt(0).toUpperCase()}
   398→              </div>
   399→              <div className="min-w-0 flex-1">
   400→                <div className="flex flex-wrap items-baseline gap-x-2 gap-y-0.5">
   401→                  <p className="text-sm font-semibold leading-tight">
   402→                    {email.fromName ?? email.fromEmail}
   403→                  </p>
   404→                  <p className="truncate text-xs text-muted-foreground">&lt;{email.fromEmail}&gt;</p>
   405→                  {isInstitutional && (
   406→                    <Badge
   407→                      variant="outline"
   408→                      className="shrink-0 gap-1 border-success/30 bg-success/10 px-1.5 py-0 text-[10px] font-medium text-success"
   409→                    >
   410→                      <ShieldCheck className="h-3 w-3" />
   411→                      Verified · {domain}
   412→                    </Badge>
   413→                  )}
   414→                </div>
   415→                <div className="mt-1.5 flex flex-wrap items-center gap-1.5">
   416→                  <span className="inline-flex items-center gap-1 text-[11px] text-muted-foreground">
   417→                    <AtSign className="h-3 w-3" /> to
   418→                  </span>
   419→                  {email.toRecipients.slice(0, 3).map((r, i) => (
   420→                    <Badge
   421→                      key={`${r.email}-${i}`}
   422→                      variant="outline"
   423→                      className="max-w-[14rem] truncate px-1.5 py-0 text-[11px] font-normal text-muted-foreground"
   424→                    >
   425→                      <span className="truncate">{r.name ?? r.email}</span>
   426→                    </Badge>
   427→                  ))}
   428→                  {email.toRecipients.length > 3 && (
   429→                    <Badge variant="outline" className="px-1.5 py-0 text-[11px] text-muted-foreground">
   430→                      +{email.toRecipients.length - 3}
   431→                    </Badge>
   432→                  )}
   433→                </div>
   434→                <div className="mt-1.5 flex flex-wrap items-center gap-2 text-[11px] text-muted-foreground">
   435→                  <span className="font-medium text-foreground/80">
   436→                    {formatRelative(email.receivedAt)}
   437→                  </span>
   438→                  <span aria-hidden>·</span>
   439→                  <span>{formatDateTime(email.receivedAt)}</span>
   440→                </div>
   441→              </div>
   442→            </div>
   443→          </div>
   444→
   445→          <SleekSeparator className="mb-5" />
   446→
   447→          {/* Thread context panel */}
   448→          {email.threadId && others.length > 0 && (
   449→            <Collapsible
   450→              open={threadOpen}
   451→              onOpenChange={setThreadOpen}
   452→              className="mb-5 overflow-hidden rounded-lg border bg-muted/30"
   453→            >
   454→              <CollapsibleTrigger asChild>

   456→                  <ChevronDown
   457→                    className={cn(
   458→                      'h-3.5 w-3.5 transition-transform',
   459→                      threadOpen ? 'rotate-180' : 'rotate-0',

   487→                            <p className="truncate text-xs font-medium">
   488→                              {m.subject ?? '(no subject)'}

   506→          {/* Body */}

   512→              </span>
   513→            </div>
   514→            <div className="mx-auto max-w-2xl overflow-x-auto">
   515→              {email.bodyHtmlSanitized ? (
   516→                <div
   517→                  className={cn(
   518→                    'min-w-0 text-[15px] leading-7 text-foreground/90',
   519→                    '[&_p]:my-3 [&_p:first-child]:mt-0 [&_p:last-child]:mb-0',
   520→                    '[&_a]:text-primary [&_a]:underline [&_a]:underline-offset-2 hover:[&_a]:opacity-80',
   521→                    '[&_strong]:font-semibold [&_em]:italic',
   522→                    '[&_ul]:my-3 [&_ul]:list-disc [&_ul]:pl-5',

   768→// Attachment card v2 — colored left accent by file type, file-type icon,
   769→// filename + size, Preview button (opens Dialog) + download icon (cosmetic).
   770→// ---------------------------------------------------------------------------
   771→type FileKind = 'pdf' | 'xls' | 'doc' | 'img' | 'ppt' | 'default'
   772→
   773→const FILE_KIND_TINT: Record<FileKind, string> = {
   774→  pdf: 'cat-red',
   775→  xls: 'cat-green',
   776→  doc: 'cat-teal',
   777→  img: 'cat-violet',
   778→  ppt: 'cat-orange',
   779→  default: 'cat-slate',
   780→}
   781→
   782→function fileKind(mimeType: string): FileKind {
   783→  const m = mimeType.toLowerCase()
   784→  if (m.includes('pdf')) return 'pdf'
   785→  if (m.includes('sheet') || m.includes('excel') || m.includes('xls') || m.includes('csv')) return 'xls'
   786→  if (m.includes('word') || m.includes('document') || m.includes('doc')) return 'doc'
   787→  if (m.startsWith('image/')) return 'img'
   788→  if (m.includes('presentation') || m.includes('powerpoint') || m.includes('ppt')) return 'ppt'
   789→  return 'default'
   790→}

   949→// ---------------------------------------------------------------------------

   952→function linkLetter(url: string): string {
   953→  const host = safeHost(url)
   954→  return host ? host.charAt(0).toUpperCase() : '#'
   955→}
   956→
   957→function safeHost(url: string): string {
   958→  try {
   959→    return new URL(url).hostname.replace(/^www\./, '')
   960→  } catch {
   961→    return url
   962→  }
   963→}
   964→
   965→function truncatePath(url: string): string {
   966→  try {
   967→    const u = new URL(url)
   968→    const path = u.pathname.replace(/^\//, '')
   969→    return path.length > 42 ? path.slice(0, 42) + '…' : path || '(root)'
   970→  } catch {
   971→    const truncated = url.length > 42 ? url.slice(0, 42) + '…' : url
   972→    return truncated
   973→  }
   974→}
