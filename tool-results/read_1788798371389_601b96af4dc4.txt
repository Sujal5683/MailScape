    80→const OKLCH_COLOR_MAP: Record<string, string> = {
    81→  amber: 'oklch(0.75 0.16 75)',
    82→  blue: 'oklch(0.62 0.17 245)',
    83→  violet: 'oklch(0.62 0.2 300)',
    84→  teal: 'oklch(0.65 0.13 180)',
    85→  rose: 'oklch(0.65 0.2 15)',
    86→  red: 'oklch(0.62 0.22 25)',
    87→  orange: 'oklch(0.7 0.17 55)',
    88→  fuchsia: 'oklch(0.65 0.22 330)',
    89→  green: 'oklch(0.65 0.16 150)',
    90→  slate: 'oklch(0.6 0.02 250)',
    91→}
    92→
    93→function oklchFor(color: string | null | undefined): string {
    94→  if (!color) return OKLCH_COLOR_MAP.slate!
    95→  return OKLCH_COLOR_MAP[color] ?? OKLCH_COLOR_MAP.slate!
    96→}

   102→type Tone = 'default' | 'warning' | 'destructive' | 'success'

   104→function toneTextClass(tone: Tone): string {
   105→  switch (tone) {
   106→    case 'warning':
   107→      return 'text-warning'
   108→    case 'destructive':
   109→      return 'text-destructive'
   110→    case 'success':
   111→      return 'text-success'
   112→    default:
   113→      return 'text-foreground'
   114→  }
   115→}
   116→
   117→function toneBadgeClass(tone: Tone): string {
   118→  switch (tone) {
   119→    case 'warning':
   120→      return 'bg-warning/15 text-warning border-warning/30'
   121→    case 'destructive':
   122→      return 'bg-destructive/15 text-destructive border-destructive/30'
   123→    case 'success':
   124→      return 'bg-success/15 text-success border-success/30'
   125→    default:
   126→      return 'bg-muted text-muted-foreground border-border'
   127→  }

   143→/** Parse a YYYY-MM-DD trend date label into a short "Mon D" string. */
   144→function formatTrendDate(iso: string): string {
   145→  // Append T00:00:00 to avoid UTC-shifted day display.
   146→  const d = new Date(`${iso}T00:00:00`)
   147→  if (Number.isNaN(d.getTime())) return iso
   148→  return d.toLocaleDateString(undefined, { month: 'short', day: 'numeric' })
   149→}
   150→
   151→function getInitial(name: string | null, email: string): string {
   152→  const src = (name ?? email).trim()
   153→  return src ? src[0]!.toUpperCase() : '?'
   154→}

   158→  if (!data) return new Map()
   159→  return new Map(data.categoryCounts.map((c) => [c.id, c.color]))

   178→        show: {
   179→          opacity: 1,
   180→          transition: { staggerChildren: prefersReducedMotion ? 0 : 0.05 },
   181→        },
   182→      },
   183→      item: {
   184→        hidden: { opacity: prefersReducedMotion ? 1 : 0, y: prefersReducedMotion ? 0 : 8 },
   185→        show: {
   186→          opacity: 1,
   187→          y: 0,
   188→          transition: { duration: 0.3, ease: 'easeOut' as const },
   189→        },
   190→      },
   191→    }),

   453→function KpiCard({ def }: { def: KpiDef }) {
   454→  const Icon = def.icon
   455→  return (
   456→    <Card
   457→      className={cn(
   458→        'gap-0 py-4 transition-transform hover:-translate-y-0.5',
   459→        def.onClick && 'cursor-pointer',
   460→      )}
   461→      onClick={def.onClick}
   462→      role={def.onClick ? 'button' : undefined}
   463→      tabIndex={def.onClick ? 0 : undefined}
   464→      onKeyDown={(e) => {
   465→        if (def.onClick && (e.key === 'Enter' || e.key === ' ')) {
   466→          e.preventDefault()
   467→          def.onClick()
   468→        }
   469→      }}
   470→    >
   471→      <CardContent className="space-y-2 px-4">
   472→        <div className="flex items-center justify-between gap-2">
   473→          <span className="truncate text-xs font-medium text-muted-foreground">
   474→            {def.label}
   475→          </span>
   476→          <span
   477→            className={cn(
   478→              'flex h-7 w-7 shrink-0 items-center justify-center rounded-full',
   479→              toneIconWrapClass(def.tone),
   480→            )}
   481→          >
   482→            <Icon className="h-3.5 w-3.5" />
   483→          </span>
   484→        </div>
   485→        <div className={cn('text-2xl font-semibold tabular-nums leading-none', toneTextClass(def.tone))}>
   486→          {def.value.toLocaleString()}
   487→        </div>
   488→        <div className="flex items-center justify-between gap-2">
   489→          <span className="truncate text-[11px] text-muted-foreground">{def.context}</span>
   490→          {def.spark && def.spark.length > 1 ? (
   491→            <Sparkline data={def.spark} />
   492→          ) : def.delta ? (
   493→            <span className="shrink-0 text-[11px] font-medium text-muted-foreground">
   494→              {def.delta}
   495→            </span>
   496→          ) : null}
   497→        </div>
   498→      </CardContent>
   499→    </Card>
   500→  )
   501→}

   800→function TrendCard({ trend }: { trend: DashboardData['trend'] }) {

   804→
   805→  // Delta: last 7 days vs first 7 days (if trend has 14 entries).
   806→  const lastHalf = trend.slice(Math.max(0, trend.length - 7)).reduce((acc, d) => acc + d.count, 0)
   807→  const firstHalf = trend.slice(0, Math.max(0, trend.length - 7)).reduce((acc, d) => acc + d.count, 0)

   809→    firstHalf === 0
   810→      ? lastHalf > 0

  1112→function deadlineTone(d: Deadline): Tone {
  1113→  const days = daysUntil(d.dueAt)
  1114→  if (days === null) return 'default'
  1115→  if (days < 0) return 'destructive'
  1116→  if (days <= 2) return 'warning'
  1117→  return 'default'
  1118→}
  1119→
  1120→function deadlineLabel(d: Deadline): string {
  1121→  const days = daysUntil(d.dueAt)
  1122→  if (days === null) return 'No due date'
  1123→  if (days < 0) return `${Math.abs(days)}d overdue`
  1124→  if (days === 0) return 'Due today'
  1125→  if (days === 1) return 'Due tomorrow'
  1126→  return `In ${days}d`
